1. OVERVIEW BASIS DATA SPASIAL
Basis data sistem  SIGAP-TRANS KALSEL dirancang untuk mendukung penyimpanan data tabular kependudukan historis (1953–
2025), data spasial titik koordinat centroid UPT, batas poligon desa/konsesi, penanganan alur verifikasi draf pengajuan kabupaten
(approval workflow), e-arsip berkas Berita Acara Serah Terima (BAST), serta audit jejak operasional pimpinan daerah. 
Prinsip Perancangan Database:
Integritas Spasial Standar OGC: Memanfaatkan tipe data geometri native PostGIS dengan format GEOMETRY(Point, 4326) untuk
titik centroid dan GEOMETRY(MultiPolygon, 4326) untuk batas deliniasi wilayah.
Indeks Spasial GIST: Memaksimalkan kecepatan pencarian lokasi, filter bounding-box (BBOX), dan clustering peta pada Leaflet.js
dengan indeks GiST (Generalized Search Tree).
Auditability & Workflow: Memisahkan antara data aktif yang tampil di publik (tabel upt_locations) dengan draf perubahan yang
diajukan oleh operator kabupaten (tabel upt_change_requests).

4. OPTIMASI & INDEKS SPASIAL (SPATIAL INDEXING)
Untuk menjaga performa rendering peta interaktif se-Kalsel agar pemuatan data berlangsung di bawah 2,5 detik, dibuat indeks spasial
dan indeks B-Tree strategis: 
Indeks Spasial GIST pada coordinate_point : Mempercepat komputasi kalkulasi jarak, pencarian radius (*ST_DWithin*), dan
*clustering* marker di Leaflet.js.
Indeks Spasial GIST pada polygon_area : Mempercepat pengujian irisan (*ST_Intersects*) antara batas desa UPT dengan
kawasan hutan lindung / IUP tambang.
Indeks B-Tree Komposit pada regency_id  & issue_status : Mengoptimalkan kecepatan panel filter cepat di dashboard
eksekutif saat menyaring status pin merah atau kabupaten tertentu.

-- 1. AKTIFKAN EKSTENSI SPASIAL POSTGIS & UUID
CREATE EXTENSION IF NOT EXISTS postgis;
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
-- 2. TABEL REGENCIES (MASTER 9 KABUPATEN)
CREATE TABLE regencies (
    id SERIAL PRIMARY KEY,
    code_roman VARCHAR(5) NOT NULL UNIQUE,
    name VARCHAR(100) NOT NULL,
    capital_city VARCHAR(100) NOT NULL,
    boundary_polygon GEOMETRY(MultiPolygon, 4326),
    created_at TIMESTAMP NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMP NOT NULL DEFAULT NOW()
);
-- 3. TABEL USERS (HAK AKSES / RBAC)
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    nip VARCHAR(30) UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    role VARCHAR(30) NOT NULL CHECK (role IN ('super_admin', 'operator_kabupaten', 'eksekutif', 'mitra_bpn')),
    regency_id INT REFERENCES regencies(id) ON DELETE SET NULL,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMP NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMP NOT NULL DEFAULT NOW()
);
-- 4. TABEL UPT_LOCATIONS (MASTER 124 UPT KALSEL)
CREATE TABLE upt_locations (
    id BIGSERIAL PRIMARY KEY,
    regency_id INT NOT NULL REFERENCES regencies(id) ON DELETE RESTRICT,
    upt_number INT NOT NULL UNIQUE CHECK (upt_number BETWEEN 1 AND 124),
    upt_name VARCHAR(150) NOT NULL,
    current_village_name VARCHAR(150) NOT NULL,
    business_pattern VARCHAR(50) NOT NULL,
    placement_year VARCHAR(20) NOT NULL,
    placement_kk INT NOT NULL DEFAULT 0,
    placement_population INT NOT NULL DEFAULT 0,
    handover_year VARCHAR(20),
    handover_kk INT NOT NULL DEFAULT 0,
    handover_population INT NOT NULL DEFAULT 0,
    issue_status VARCHAR(20) NOT NULL DEFAULT 'clean' CHECK (issue_status IN ('clean', 'warning', 'critical')),
    issue_note TEXT,
    coordinate_point GEOMETRY(Point, 4326) NOT NULL,
    polygon_area GEOMETRY(MultiPolygon, 4326),
    is_verified BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMP NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMP NOT NULL DEFAULT NOW()
);
-- INDEKS SPASIAL & SEARCH OPTIMIZATION
CREATE INDEX idx_upt_point_gist ON upt_locations USING GIST (coordinate_point);
CREATE INDEX idx_upt_polygon_gist ON upt_locations USING GIST (polygon_area);
CREATE INDEX idx_upt_regency_status ON upt_locations (regency_id, issue_status);
-- 5. TABEL UPT_CHANGE_REQUESTS (DRAF VERIFIKASI)
CREATE TABLE upt_change_requests (
    id BIGSERIAL PRIMARY KEY,
    upt_location_id BIGINT NOT NULL REFERENCES upt_locations(id) ON DELETE CASCADE,
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    request_type VARCHAR(50) NOT NULL,
    proposed_payload JSONB NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'approved', 'rejected')),
    reviewer_note TEXT,
    reviewed_by UUID REFERENCES users(id) ON DELETE SET NULL,
    reviewed_at TIMESTAMP,
    created_at TIMESTAMP NOT NULL DEFAULT NOW()
);
-- 6. TABEL UPT_DOCUMENTS (E-ARSIP BAST)
CREATE TABLE upt_documents (
    id BIGSERIAL PRIMARY KEY,
    upt_location_id BIGINT NOT NULL REFERENCES upt_locations(id) ON DELETE CASCADE,
    document_type VARCHAR(50) NOT NULL CHECK (document_type IN ('BAST', 'SK_GUBERNUR', 'SK_MENTERI', 'BUKU_TANAH')),
    document_number VARCHAR(100),
    file_name VARCHAR(255) NOT NULL,
    file_path VARCHAR(255) NOT NULL,
    file_size_kb INT NOT NULL,
    uploaded_by UUID NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
    created_at TIMESTAMP NOT NULL DEFAULT NOW()
);
